cmake_minimum_required(VERSION 3.20)
project(parallel_stats VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(parallel_stats INTERFACE)
target_include_directories(parallel_stats INTERFACE include)
target_link_libraries(parallel_stats INTERFACE Threads::Threads)
if(MSVC)
  target_compile_options(parallel_stats INTERFACE /W4 /WX /permissive- /EHsc)
else()
  target_compile_options(parallel_stats INTERFACE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(stats-tests tests/tests.cpp)
target_link_libraries(stats-tests PRIVATE parallel_stats)
add_executable(stats-benchmark src/benchmark.cpp)
target_link_libraries(stats-benchmark PRIVATE parallel_stats)
enable_testing()
foreach(name bounds values exception drain closed blocked_close backpressure reentrant producers moments determinism nonfinite stress)
  add_test(NAME ${name} COMMAND stats-tests ${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 30)
endforeach()
